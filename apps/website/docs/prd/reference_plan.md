Research & Reference System

Description
We want a feature which tracks references (online resources in the first instances). These can be added directly by the user in the UI, or via a form or via another process. The initial input will consist of a URL and optionally a short description.

There are two types of references. Canonical (primary) references which are entered into the system and secondary references, which are links found in canonical references.

Canonical references will be automatically enriched via an n8n workflow which uses Tavily for this purpose. URLs which are found in references will be stored as secondary references, which have a reference count (i.e. the number of canonical references which link to them). If the reference count is > 1 for a reference and it isn't canonical it will itself be promoted to canonical and scheduled for enrichment.

The enrichment process will be as follows:

- a trigger attached to the reference will schedule a new reference (or one which has been changed to canonical = true and hasn't already been enriched) to a Supabase Queue

A Supabase Edge Function will be called from a cron job every n seconds. It will inspect the queue and if there is an item available in it then it will send it to an n8n workflow for enrichment. The item in the queue will be marked as not available to be sent again for a period of time.

Another edge function will be invoked via a web hook from n8n when the processing of the reference enrichment has finished (successfully, unsuccessfully or timed out.). Depending upon the outcome it will remove the item from the queue or leave it there to be attempted later. There needs to be a mechanism to remove the item after a certain number of failed attempts to enrich it.

The n8n function will have a built-in timeout functionality and so should always return either an enriched record, an error or a timeout indication.

We need to ensure that there is no part of this feature or its processes which could lead to runaway enrichment (which could get expensive in terms of Tavily's API)

Later on we may add additional features such as a ranking of the article of some kind.

-- AFTER UPDATE trigger on public.reference
-- Ensures outbound reference links are synchronized, reference counts are correct,
-- and canonical/promotion/enrichment logic is enforced.
-- All link management is handled in manage_reference_links using both OLD and NEW linked_urls.

CREATE OR REPLACE FUNCTION public.tr_reference_u_a()
RETURNS TRIGGER AS $$
BEGIN
-- Enqueue for enrichment if process flips from false to true
IF (NOT OLD.process) AND NEW.process THEN
PERFORM pgmq_public.send('reference_expander', jsonb_build_object('reference_id', NEW.id), 0);
END IF;

-- Update all link relationships and reference counts
PERFORM public.manage_reference_links(
NEW.id,
NEW.brand,
OLD.linked_urls,
NEW.linked_urls
);

RETURN NEW;
END;

$$
LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS tr_reference_u_a ON public.reference;
CREATE TRIGGER tr_reference_u_a
AFTER UPDATE ON public.reference
FOR EACH ROW

-- AFTER INSERT trigger on public.reference

-- ============================================================================
-- FUNCTION: manage_reference_links
-- Maintains reference linkage and counts for the reference table.
--
-- Called from AFTER UPDATE/INSERT triggers on reference.
-- - Removes links that no longer exist in linked_urls.
-- - Adds new links as necessary, creating child references if needed.
-- - Updates reference_count for all affected references.
-- - Promotes a child to canonical/process if referenced by >1 canonical records,
--   but only if not already enriched.
--
-- This function expects both old and new linked_urls as jsonb[] arrays.
-- ============================================================================
CREATE OR REPLACE FUNCTION public.manage_reference_links(
  reference_id bigint,
  brand_id bigint,
  old_linked_urls jsonb[],
  new_linked_urls jsonb[]
) RETURNS void AS
$$

DECLARE
old_urls text[] := '{}';
new_urls text[] := '{}';
url text;
target_id bigint;
BEGIN
-- Parse old linked_urls to array
IF old_linked_urls IS NOT NULL THEN
old_urls := ARRAY(
SELECT linked->>'url'
FROM unnest(old_linked_urls) AS linked
WHERE linked ? 'url' AND linked->>'url' IS NOT NULL
);
END IF;

-- Parse new linked_urls to array
IF new_linked_urls IS NOT NULL THEN
new_urls := ARRAY(
SELECT linked->>'url'
FROM unnest(new_linked_urls) AS linked
WHERE linked ? 'url' AND linked->>'url' IS NOT NULL
);
END IF;

-- 1. Remove links in OLD but not in NEW
FOREACH url IN ARRAY old_urls LOOP
IF NOT url = ANY(new_urls) THEN
-- Remove the link
DELETE FROM reference_reference
WHERE source = reference_id AND target = (SELECT id FROM reference WHERE url = url);

      -- Recompute reference_count for the target
      UPDATE reference SET reference_count = (
        SELECT COUNT(*) FROM reference_reference WHERE target = id
      ) WHERE url = url;
      -- Optionally: Flag orphaned refs for cleanup/reporting elsewhere
    END IF;

END LOOP;

-- 2. Add links in NEW but not in OLD
FOREACH url IN ARRAY new_urls LOOP
IF NOT url = ANY(old_urls) THEN
-- Upsert target reference
INSERT INTO reference (url, brand)
VALUES (url, brand_id)
ON CONFLICT (url) DO NOTHING;

      -- Get target id
      SELECT id INTO target_id FROM reference WHERE url = url LIMIT 1;

      -- Insert reference_reference
      INSERT INTO reference_reference (source, target)
        VALUES (reference_id, target_id)
      ON CONFLICT DO NOTHING;

      -- Update reference_count for target
      UPDATE reference SET reference_count = (
        SELECT COUNT(*) FROM reference_reference WHERE target = id
      ) WHERE id = target_id;

      -- Promote if needed: Only promote if not canonical, count > 1, and not yet enriched
      UPDATE reference SET
        canonical = TRUE,
        process = CASE
          WHEN process = FALSE AND (raw_content IS NULL OR LENGTH(TRIM(raw_content)) = 0) THEN TRUE
          ELSE process
        END
      WHERE id = target_id
        AND canonical = FALSE
        AND reference_count > 1;
    END IF;

END LOOP;
END;

$$
LANGUAGE plpgsql;

// Edge Function: Reliable queue processing using pgmq_read for at-least-once delivery
// Pops a job, processes it, and only deletes after success. Retries on failure.
import "jsr:@supabase/functions-js/edge-runtime.d.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";
const N8N_WEBHOOK_URL = "https://n8n.novansa.com/webhook/enrich-reference";
function getServiceClient() {
  // @ts-ignore
  return createClient(Deno.env.get("SUPABASE_URL") ?? "", Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") ?? "");
}
Deno.serve(async (_req)=>{
  const supabase = getServiceClient();
  // Log helper
  async function log(message, json = null) {
    try {
      await supabase.from("dbg_log").insert([
        {
          message,
          json
        }
      ]);
    } catch (_) {
    // suppress logging errors
    }
  }
  // 1. Read (reserve) job from managed queue
  const { data: readResult, error: readError } = await supabase.rpc("pgmq_read", {
    queue_name: "reference_expander",
    sleep_seconds: 30,
    n: 1
  });
  const popped = Array.isArray(readResult) && readResult.length > 0 ? readResult[0] : null;
  if (readError || !popped || !popped.message) {
    await log("No pending jobs or error during pgmq_read", {
      readError,
      readResult
    });
    return new Response(JSON.stringify({
      message: "No pending jobs"
    }), {
      status: 200
    });
  }
  await log("Dequeued job", popped);
  // 2. Get reference record
  const referenceId = popped.message.reference_id;
  const { data: ref, error: refErr } = await supabase.from("reference").select("*").eq("id", referenceId).single();
  if (refErr || !ref) {
    await log("Reference not found", {
      referenceId,
      refErr
    });
    return new Response(JSON.stringify({
      message: "Reference not found"
    }), {
      status: 404
    });
  }
  await log("Reference fetched", {
    referenceId,
    ref
  });
  // 3. POST to n8n webhook
  let n8nResp, n8nRespJson;
  const controller = new AbortController();
  const timeout = setTimeout(()=>controller.abort(), 30000);
  try {
    n8nResp = await fetch(N8N_WEBHOOK_URL, {
      method: "POST",
      headers: {
        "Content-Type": "application/json"
      },
      body: JSON.stringify(ref),
      signal: controller.signal
    });
    clearTimeout(timeout);
  } catch (err) {
    clearTimeout(timeout);
    await log("n8n webhook fetch error", {
      error: err.toString(),
      referenceId
    });
    // Do NOT delete the message; it will be retried after timeout
    return new Response(JSON.stringify({
      message: `Webhook error: ${err}`
    }), {
      status: 502
    });
  }
  if (n8nResp.ok) {
    try {
      n8nRespJson = await n8nResp.json();
      await log("n8n response parsed", {
        response: n8nRespJson
      });
      if (typeof n8nRespJson === "object" && n8nRespJson.id === ref.id) {
        // Patch fields back to reference as needed
        const { id, created_at, updated_at, ...updatableFields } = n8nRespJson;
        await supabase.from("reference").update({
          ...updatableFields,
          updated_at: new Date().toISOString()
        }).eq("id", ref.id);
        await log("Reference updated", {
          referenceId: ref.id,
          updates: updatableFields
        });
        // 5. Delete processed queue message
        await supabase.rpc("pgmq_delete", {
          queue_name: "reference_expander",
          message_id: popped.message_id
        });
        await log("Queue item deleted", {
          referenceId,
          messageId: popped.message_id
        });
        return new Response(JSON.stringify({
          message: "Success",
          n8nResponse: n8nRespJson
        }), {
          status: 200
        });
      } else {
        await log("Invalid n8n response (ID mismatch)", {
          expected: ref.id,
          received: n8nRespJson
        });
        // Do NOT delete the message; it will be retried after timeout
        return new Response(JSON.stringify({
          message: "Invalid response from n8n"
        }), {
          status: 500
        });
      }
    } catch (err) {
      await log("n8n JSON parse error", {
        error: err.toString()
      });
      // Do NOT delete the message; it will be retried after timeout
      return new Response(JSON.stringify({
        message: `JSON parse error: ${err}`
      }), {
        status: 500
      });
    }
  } else {
    const errText = await n8nResp.text();
    await log("n8n HTTP error", {
      status: n8nResp.status,
      response: errText
    });
    // Do NOT delete the message; it will be retried after timeout
    return new Response(JSON.stringify({
      message: "n8n error",
      n8nResponse: errText
    }), {
      status: 500
    });
  }
});
